\chapter{Minimal hypersurfaces and stability}\label{sec:minimal-stability}
In the variational chapters we write $M^{n+1}$ for the ambient manifold and $\Sigma^n$ for a hypersurface. Thus $n$ is the hypersurface dimension. A surface has $n=2$. These conventions replace the Euclidean dimension letters used in the foundational chapters. The smooth theory connects the first variation of varifolds with elliptic equations and Morse index. Sources for this section are Marques' lectures, recorded by Montezuma, \cite{MM}, Sections 2 and 6, and Chodosh \cite{ChMin}, Sections 7 and 23.

\begin{definition}[Mean curvature and minimality]\label{def:smooth-minimal}
For a smooth immersed hypersurface $\Sigma\subset M$, let
\[
 B(X,Y)=(\nabla^M_XY)^\perp,\qquad
 \vec H=\operatorname{tr}_{T\Sigma}B.
\]
We call $\Sigma$ minimal when $\vec H=0$. If a global unit normal $\nu$ exists, $\Sigma$ is two-sided. Our scalar convention is $H=\operatorname{div}_{\Sigma}\nu$, so that $\vec H=-H\nu$. With the outward normal, a Euclidean sphere of radius $R$ has $H=n/R$. Here $H$ is the sum of principal curvatures; in the Willmore energy we explicitly use their average. See \cite{MM}, Section 2, and \cite{ChMin}, Section 7.
\end{definition}

\begin{proposition}[First variation of a closed hypersurface]\label{prop:smooth-first-variation}
Let $\Sigma$ be smooth, compact, and without boundary. If $F_t$ is a smooth variation with velocity $X$, then
\[
 \left.\frac{d}{dt}\right|_{0}\operatorname{Area}(F_t(\Sigma))
 =\int_\Sigma\operatorname{div}_\Sigma X\,dA
 =-\int_\Sigma\langle\vec H,X\rangle\,dA.
\]
For $X=f\nu$ this equals $\int_\Sigma Hf\,dA$. This is the smooth counterpart of the varifold first variation. Compare \cite{MM}, Theorems 2.3 and 2.6.
\end{proposition}
\begin{proof}
In local coordinates, differentiating the induced metric gives
$\dot g_{ij}=\langle\nabla_iX,e_j\rangle+\langle e_i,\nabla_jX\rangle$.
The identity $\frac{d}{dt}\sqrt{\det g}=\frac12\operatorname{tr}(g^{-1}\dot g)\sqrt{\det g}$ therefore gives $\dot{dA}=\operatorname{div}_\Sigma X\,dA$.
Decompose $X=X^T+X^\perp$. Differentiating $\langle X^\perp,e_i\rangle=0$ along an orthonormal tangent frame yields
$\operatorname{div}_\Sigma X=\operatorname{div}_\Sigma X^T-\langle\vec H,X\rangle$.
The tangential divergence integrates to zero because $\Sigma$ has no boundary. Finally substitute $\vec H=-H\nu$.
\end{proof}

\begin{theorem}[Second variation in the two-sided case]\label{thm:second-variation}
Let $\Sigma^n\subset(M^{n+1},g)$ be a smooth, closed, two-sided minimal hypersurface. For a smooth normal variation with initial velocity $f\nu$, the second derivative of area is
\[
 Q(f,f)=\int_\Sigma\bigl(|\nabla_\Sigma f|^2-(|B|^2+\operatorname{Ric}_M(\nu,\nu))f^2\bigr)\,dA.
\]
With $\Delta_\Sigma=\operatorname{div}_\Sigma\nabla_\Sigma$, define
$J=\Delta_\Sigma+|B|^2+\operatorname{Ric}_M(\nu,\nu)$; then $Q(f,f)=-\int_\Sigma fJf\,dA$.
Source: \cite{MM}, Theorem 6.2, specialized to a closed hypersurface; also \cite{ChMin}, Section 23.
\end{theorem}
\begin{proof}
Differentiate the first-variation formula $\frac d{dt}A(\Sigma_t)=\int_{\Sigma_t}H_tf_t\,dA_t$. At a minimal initial hypersurface, the terms from differentiating $f_t$ and $dA_t$ vanish because $H_0=0$. Thus $A''(0)=\int_\Sigma f\dot H$. The Riemannian calculation in Proposition \ref{prop:riemannian-mean-curvature-variation}, which does not use the second variation, gives $\dot H=-\Delta_\Sigma f-(|B|^2+\operatorname{Ric}(\nu,\nu))f$. Integration by parts on the closed hypersurface proves the formula. A tangential initial velocity contributes the tangential derivative of $H_0=0$, so only the normal speed enters. The same proof works for variations compactly supported away from a boundary.
\end{proof}

\begin{definition}[Stability and Morse index]\label{def:morse-index}
For the hypersurface in Theorem \ref{thm:second-variation}, stability means $Q(f,f)\geq0$ for every smooth $f$. The Morse index is the largest dimension of a linear subspace on which $Q$ is negative definite. A Jacobi field is a normal field in the kernel of the Jacobi operator. For a one-sided hypersurface, use sections of its normal line bundle, equivalently the appropriate equivariant functions on the two-sided cover. These are unrestricted area variations; CMC stability will use a volume constraint. See \cite{MM}, Section 9, and \cite{MNIndex}, Section 1.
\end{definition}

\begin{corollary}[Positive Ricci curvature gives an unstable direction]\label{cor:ricci-unstable}
If $\operatorname{Ric}_M>0$, every nonempty smooth closed two-sided minimal hypersurface has Morse index at least one.
\end{corollary}
\begin{proof}
Put $f=1$ in Theorem \ref{thm:second-variation}. The gradient term vanishes, while $|B|^2+\operatorname{Ric}_M(\nu,\nu)>0$ everywhere. Consequently $Q(1,1)<0$, and the span of the constant function is a negative direction.
\end{proof}

\chapter{Minimal graphs and elliptic regularity}\label{sec:minimal-graphs}
The graph equation is a first bridge from geometric variation to nonlinear elliptic PDE. Simon's supplied \emph{Lectures on PDE}, last updated March 5, 2015, develops the analytic background and treats the minimal surface and mean curvature equations in Chapter 19 \cite{SimonPDE}.

\begin{proposition}[The minimal graph equation]\label{prop:minimal-graph-pde}
Let $U\subset\mathbb R^n$ be open and $u\in C^2(U)$. The graph of $u$ is stationary for compactly supported vertical variations if and only if
\[
 \operatorname{div}\left(\frac{Du}{\sqrt{1+|Du|^2}}\right)=0.
\]
Equivalently,
$\bigl(\delta_{ij}-u_i u_j/(1+|Du|^2)\bigr)u_{ij}=0$,
where repeated indices are summed. Compare \cite{SimonPDE}, Chapter 19, and \cite{ChMin}, Section 18.
\end{proposition}
\begin{proof}
For a relatively compact domain containing the support of a test function $\phi$, the graph area is $\int\sqrt{1+|Du|^2}$ by the area formula. Its derivative at $u+t\phi$ is
$\int Du\cdot D\phi/\sqrt{1+|Du|^2}$.
Integration by parts gives the stated equation. Expanding the divergence gives the nondivergence expression. Since $u$ is $C^2$, the weak and pointwise expressions agree.
\end{proof}

\begin{remark}[Regularity techniques and their hypotheses]\label{rem:regularity-program}
The analytic and geometric regularity inputs are developed separately below. Definitions \ref{def:weak-minimal-graph} and \ref{def:weak-elliptic-equation} specify weak equations; Proposition \ref{prop:graph-ellipticity} identifies the gradient bound that gives uniform ellipticity. Theorems \ref{thm:elliptic-harnack} and \ref{thm:interior-schauder} provide distinct estimates for measurable and Hölder coefficients. The graph comparison and calibration results then give geometric consequences.

For nonsmooth varifolds, Theorem \ref{thm:tangent-cones} constructs blow-ups, Theorem \ref{thm:allard-lp} states the density and mean-curvature hypotheses for Allard regularity, and Theorem \ref{thm:stable-curvature-bound} gives a low-dimensional stable curvature estimate. Their sources are \cite{SimonPDE}, \cite{ChMin}, and \cite{Minter}. These results do not assert unrestricted regularity for stationary varifolds; cylindrical-cone and general sheeting theories still require their own development.
\end{remark}

\begin{definition}[Weak minimal graphs]\label{def:weak-minimal-graph}
A function $u\in W^{1,\infty}_{\mathrm{loc}}(U)$, $U\subset\mathbb R^n$ open, is a weak solution of the scalar minimal surface equation if
\[
 \int_U F(Du)\cdot D\phi=0\quad(\phi\in C_c^\infty(U)),
 \qquad F(p)=\frac p{\sqrt{1+|p|^2}}.
\]
The integrand is locally integrable, so the equation is defined without second derivatives. Here the graph has codimension one. See \cite{Minter}, Section 3.1, p.~48.
\end{definition}

\begin{proposition}[Ellipticity and the gradient bound]\label{prop:graph-ellipticity}
For $F$ in Definition \ref{def:weak-minimal-graph}, write $W(p)=\sqrt{1+|p|^2}$. Then
\[
 A_{ij}(p)=\partial_{p_j}F_i(p)=\frac{\delta_{ij}}{W(p)}-\frac{p_i p_j}{W(p)^3}.
\]
Its eigenvalues are $W^{-1}$ on $p^\perp$ and $W^{-3}$ in the direction of $p$; at $p=0$ they are all one. Consequently, when $|p|\leq L$,
\[
 (1+L^2)^{-3/2}|\xi|^2\leq A(p)\xi\cdot\xi\leq|\xi|^2.
\]
The nondivergence coefficient matrix $a(p)=W(p)A(p)$ has eigenvalues $1$ and $(1+|p|^2)^{-1}$. These ellipticity constants deteriorate as the gradient grows. Compare \cite{ChMin}, Exercise 18.1.
\end{proposition}
\begin{proof}
Differentiate $p_i(1+|p|^2)^{-1/2}$. Decompose $\xi$ into its components parallel and perpendicular to $p$. The displayed eigenvalues and bounds follow directly.
\end{proof}

\begin{proposition}[Linearization and differences of graph solutions]\label{prop:graph-linearization}
For a smooth $u$, the linearization of $u\mapsto\operatorname{div}F(Du)$ in direction $v$ is $\operatorname{div}(A(Du)Dv)$. If $u_1,u_2$ are smooth minimal graph solutions, their difference $w=u_2-u_1$ satisfies
\[
 \operatorname{div}(\overline A Dw)=0,\qquad
 \overline A(x)=\int_0^1 A\bigl((1-t)Du_1(x)+tDu_2(x)\bigr)\,dt.
\]
If both gradients have norm at most $L$, then $\overline A$ has the bounds in Proposition \ref{prop:graph-ellipticity}. In particular each derivative $D_k u$ of a smooth solution satisfies $\operatorname{div}(A(Du)D(D_k u))=0$.
\end{proposition}
\begin{proof}
Differentiate $F(Du+tDv)$ under the divergence. For two solutions, the fundamental theorem of calculus gives $F(Du_2)-F(Du_1)=\overline A Dw$; take its divergence. The segment between the two gradients stays in the ball of radius $L$, so averaging preserves the ellipticity bounds. Differentiating the equation in $x_k$ gives the last claim. This is the divergence-form counterpart of the linearization in \cite{ChMin}, Proposition 19.3.
\end{proof}

\begin{theorem}[Smoothness of Lipschitz weak minimal graphs]\label{thm:lipschitz-minimal-regularity}
Every locally Lipschitz weak solution of the scalar minimal surface equation on an open $U\subset\mathbb R^n$ has a smooth representative in $U$. No smallness of its Lipschitz constant is required locally. This scalar result does not extend as stated to arbitrary higher-codimension minimal graph systems. The proof uses difference quotients, uniform ellipticity on compact subsets, De Giorgi--Nash--Moser regularity for derivatives, and Schauder bootstrapping. Sources: \cite{Minter}, Section 3.1, and the discussion following Theorem 2.5; \cite{SimonPDE}, Chapters 18--19. The weak-solution regularity proof remains external.
\end{theorem}

\begin{proposition}[Bootstrapping a classical minimal graph]\label{prop:classical-graph-bootstrap}
A $C^2$ solution of the minimal surface equation on $U$ belongs to $C^\infty(U)$.
\end{proposition}
\begin{proof}
On nested relatively compact balls the gradient is bounded, and the coefficients $a^{ij}(Du)$ are $C^1$, hence Hölder continuous for every exponent below one. Apply Theorem \ref{thm:interior-schauder} to $a^{ij}(Du)D_{ij}u=0$ to obtain $u\in C^{2,\alpha}_{\mathrm{loc}}$. Differentiation gives
\[
 a^{ij}(Du)D_{ij}(D_k u)+
 \bigl(\partial_{p_\ell}a^{ij}(Du)D_{ij}u\bigr)D_\ell(D_k u)=0.
\]
The lower-order coefficients are $C^{0,\alpha}$. Interior Schauder regularity gives one further derivative, with the differentiated equation justified by difference quotients if needed. Repeating on smaller balls proves the claim. This is the bootstrap in \cite{ChMin}, Lemma 19.1, pp.~51--52.
\end{proof}

\begin{proposition}[Strong comparison for minimal graphs]\label{prop:graph-strong-comparison}
Let $U$ be connected, and let $u_1,u_2\in C^2(U)$ solve the minimal surface equation. If $u_1\leq u_2$ in $U$ and equality holds at one interior point, then $u_1=u_2$ throughout $U$.
\end{proposition}
\begin{proof}
The solutions are smooth by Proposition \ref{prop:classical-graph-bootstrap}. Their nonnegative difference solves the locally uniformly elliptic equation in Proposition \ref{prop:graph-linearization}. On a sufficiently small ball about a zero, Harnack gives $\sup w\leq C\inf w=0$. Thus the zero set is open; it is also closed by continuity. Connectedness completes the proof. Compare \cite{ChMin}, Proposition 19.3.
\end{proof}

\begin{corollary}[Boundary comparison and uniqueness]\label{cor:graph-dirichlet-uniqueness}
Let $U\subset\mathbb R^n$ be a bounded connected domain. Suppose $u_1,u_2\in C^2(U)\cap C^0(\overline U)$ are minimal graph solutions and $u_1\leq u_2$ on $\partial U$. Then $u_1\leq u_2$ in $U$. In particular a classical solution with prescribed continuous boundary values is unique, when it exists.
\end{corollary}
\begin{proof}
If $t=\max_{\overline U}(u_1-u_2)>0$, the maximum is attained in $U$. The functions $u_1$ and $u_2+t$ are ordered solutions with interior contact, so Proposition \ref{prop:graph-strong-comparison} makes them equal throughout $U$. Continuity at the boundary contradicts the boundary inequality. This is the sliding argument of \cite{ChMin}, Corollary 19.4.
\end{proof}

\begin{corollary}[Interior tangency principle]\label{cor:minimal-tangency}
Let two smooth embedded minimal hypersurfaces in a smooth Riemannian manifold touch at an interior point. If one lies locally on one side of the other, then their supports agree in a neighborhood of that point.
\end{corollary}
\begin{proof}
Use a common tangent plane and a coordinate chart to write both hypersurfaces as graphs. Their difference is nonnegative and solves the linear uniformly elliptic equation obtained by integrating the linearization of the local mean-curvature operator between the graphs. The coefficients are smooth and bounded on a smaller chart. The interior Harnack principle for this equation, including its bounded lower-order coefficients, forces the difference to vanish. See \cite{ChMin}, Section 19, and \cite{SimonPDE}, Chapter 18, Theorem 1. The hypothesis is interior one-sided contact; transverse crossings do not satisfy it.
\end{proof}

\begin{definition}[Calibrations]\label{def:calibration}
On an oriented Riemannian manifold, a calibration of degree $n$ is a smooth closed $n$-form $\omega$ with comass at most one at every point. An oriented immersed $n$-submanifold $\Sigma$ is calibrated when $\omega|_{T\Sigma}=dA_\Sigma$. The comass is the supremum of the absolute value on unit simple $n$-vectors, as in Definition \ref{def:mass-current}. See \cite{ChMin}, Section 16.
\end{definition}

\begin{proposition}[A calibration proves area minimization]\label{prop:calibration-minimizes}
Let $\Sigma$ and $S$ be compact oriented smooth $n$-submanifolds with the same oriented boundary in an open set $U$, and suppose $S-\Sigma=\partial R$ for a compactly supported $(n+1)$-current $R$ in $U$. If a calibration $\omega$ on $U$ calibrates $\Sigma$, then $\operatorname{Area}(\Sigma)\leq\operatorname{Area}(S)$.
\end{proposition}
\begin{proof}
Stokes' theorem and $d\omega=0$ give $\int_S\omega-\int_\Sigma\omega=R(d\omega)=0$. A cutoff identically one near the compact supports justifies evaluating a possibly noncompactly supported form. The calibration identity gives $\int_\Sigma\omega=\operatorname{Area}(\Sigma)$, while the comass bound gives $\int_S\omega\leq\operatorname{Area}(S)$. Compare \cite{ChMin}, Theorem 16.2.
\end{proof}

\begin{proposition}[Minimal graphs are calibrated in their cylinder]\label{prop:graph-calibration}
If $u\in C^2(U)$ solves the minimal surface equation, define on $U\times\mathbb R$
\[
 N(x,z)=\frac{(-Du(x),1)}{\sqrt{1+|Du(x)|^2}},\qquad
 \omega=\iota_N(dx_1\wedge\cdots\wedge dx_n\wedge dz).
\]
Orient the graph so that $\omega$ restricts to its positive area form. Then $\omega$ calibrates it. Consequently compact portions of the graph minimize area against oriented competitors homologous to them by compactly supported currents inside $U\times\mathbb R$.
\end{proposition}
\begin{proof}
The normal field has length one, so the comass of $\omega$ is one. Also $d\omega=(\operatorname{div}N)\,dx_1\wedge\cdots\wedge dz=0$ by the graph equation and independence from $z$. On the graph, $N$ is its unit normal, giving the calibration identity. Apply Proposition \ref{prop:calibration-minimizes}. This is \cite{ChMin}, Example 16.3 and Theorem 18.1. The cylinder restriction is part of the conclusion.
\end{proof}

\begin{example}[A graph with unbounded slope at its domain boundary]\label{ex:scherk-graph}
On $(-\pi/2,\pi/2)^2$, the function $u(x,y)=\log(\cos y)-\log(\cos x)$ is a minimal graph. Indeed $u_x=\tan x$, $u_y=-\tan y$, $u_{xx}=\sec^2x$, $u_{yy}=-\sec^2y$, and $u_{xy}=0$, so
\[
 (1+u_y^2)u_{xx}-2u_xu_yu_{xy}+(1+u_x^2)u_{yy}=0.
\]
Its slope is bounded on compact subsets but diverges near the square's boundary. Thus local ellipticity does not give a single uniform ellipticity constant on the entire domain. Compare \cite{ChMin}, Exercise 18.2.
\end{example}

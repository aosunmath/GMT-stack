\chapter{Cylinders, spheres and a period obstruction}\label{sec:examples-cmc-reconstruction}
These calculations apply Section \ref{sec:cmc-reconstruction} to explicit harmonic maps. The construction is from \cite{Hel01}, pp.~19--21; the formulas below check its geometry, degeneracies, and global limitations directly.

\begin{example}[An equatorial harmonic map gives a cylinder]\label{ex:equatorial-harmonic-cylinder}
For $u(x,y)=(\cos x,\sin x,0)$ on $\mathbb R^2$, a reconstruction primitive is $B=(0,0,-y)$. The two maps
\[
 X_\varepsilon(x,y)=(\varepsilon\cos x,\varepsilon\sin x,-y)
\]
parametrize a unit circular cylinder, with induced metric $dx^2+dy^2$ and trace mean curvature $\varepsilon$ relative to $u$. They are covering immersions on $\mathbb R^2$ and descend to embedded cylinders on $(\mathbb R/2\pi\mathbb Z)\times\mathbb R$.
\end{example}
\begin{proof}
Here $u_{xx}=-u$, $u_y=0$, and $|Du|^2=1$, so $u$ is harmonic. Since $u\times u_x=(0,0,1)$, the reconstruction form is $-(0,0,1)dy=dB$. The formulas and metric follow by differentiation. The mean curvature follows from Theorem \ref{thm:cmc-harmonic-reconstruction}; for $\varepsilon=1$ it is the positive outward cylinder curvature. Periodicity in $x$ explains the covering, and the quotient map is injective onto the cylinder. The product metric is complete.
\end{proof}

\begin{example}[Harmonic phases and cylinder parametrizations]\label{ex:harmonic-phase-cylinder}
Let $\varphi$ be a smooth real harmonic function on a simply connected planar domain, and let $\psi$ be a harmonic conjugate, so $\psi_x=-\varphi_y$ and $\psi_y=\varphi_x$. Then $u=(\cos\varphi,\sin\varphi,0)$ is harmonic and
\[
 X_\varepsilon=(\varepsilon\cos\varphi,\varepsilon\sin\varphi,-\psi)
\]
is its CMC reconstruction. Its metric is $|\nabla\varphi|^2(dx^2+dy^2)$. It is an immersion exactly where $\nabla\varphi\ne0$; its image lies on the unit cylinder, but global injectivity is not implied.
\end{example}
\begin{proof}
Writing $v=(-\sin\varphi,\cos\varphi,0)$ gives $u_i=v\varphi_i$ and $\Delta u=v\Delta\varphi-u|\nabla\varphi|^2=-u|Du|^2$. Also $u\times v=e_3$, so $\alpha=e_3(\varphi_y dx-\varphi_xdy)=-e_3d\psi$. This proves the primitive and formula. The Cauchy--Riemann equations give $|X_x|^2=|X_y|^2=\varphi_x^2+\varphi_y^2$ and $X_x\cdot X_y=0$. Critical points of $\varphi$ give zero differentials; elsewhere the reconstruction theorem applies. Even without critical points, periodicity of the trigonometric functions can identify different domain points.
\end{proof}

\begin{example}[One stereographic map gives a sphere and a constant map]\label{ex:stereographic-cmc-reconstruction}
Put $r^2=x^2+y^2$ and
\[
 u(x,y)=\frac{(2x,2y,1-r^2)}{1+r^2}.
\]
This is an orientation-preserving conformal harmonic parametrization of $S^2$ minus its south pole. Its reconstruction primitive is $B=-u$, so $X_+=0$ and $X_-=-2u$. Thus one reconstruction is constant, whereas the other parametrizes a sphere of radius two with one point omitted.
\end{example}
\begin{proof}
Direct differentiation gives
\[
 |u_x|^2=|u_y|^2=\frac4{(1+r^2)^2},\quad
 u_x\cdot u_y=0,\quad u\cdot(u_x\times u_y)=\frac4{(1+r^2)^2},
 \quad\Delta u=-\frac8{(1+r^2)^2}u.
\]
The last equality is the harmonic-map equation. Orientation and conformality give $u_y=u\times u_x$. Hence $u\times u_y=-u_x$ and $-u\times u_x=-u_y$, so $\alpha=-du$. The claimed primitives and reconstructions follow. The spherical image of $-2u$ has center zero and radius two; its chosen normal $u$ points inward, so $H=-1$, consistent with Theorem \ref{thm:cmc-harmonic-reconstruction}. No extension across infinity is needed for the assertion on the planar chart.
\end{proof}

\begin{example}[A logarithmic phase with a nonzero reconstruction period]\label{ex:cmc-logarithmic-period}
On an annulus $\{a<r<b\}$ with $0<a<b<\infty$, the harmonic map
\[
 u=(\cos\log r,\sin\log r,0)
\]
has reconstruction form $\alpha=-e_3d\theta$. Its period around a positively oriented circle is $-2\pi e_3$. Consequently this reconstruction has no single-valued primitive on the annulus, although it gives cylinder parametrizations on the universal cover.
\end{example}
\begin{proof}
The function $\log r$ is harmonic away from zero. Locally its harmonic conjugate is $\theta$, so Example \ref{ex:harmonic-phase-cylinder} gives the form. The one-form $d\theta=(-y\,dx+x\,dy)/r^2$ is globally defined on the annulus, and its circle integral is $2\pi$. Proposition \ref{prop:cmc-reconstruction-periods} therefore rules out a single-valued primitive. On the cover with coordinates $(t,\theta)=(\log r,\theta)$ and real $\theta$, take $B=-\theta e_3$ and $X_\varepsilon=(\varepsilon\cos t,\varepsilon\sin t,-\theta)$. A deck transformation $\theta\mapsto\theta+2\pi$ translates the image by $-2\pi e_3$, so it does not descend as a map to $\mathbb R^3$. This is an obstruction to this construction on this domain, not a classification of all CMC immersions.
\end{proof}

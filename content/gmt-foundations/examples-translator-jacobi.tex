\chapter{Weighted Jacobi fields on grim reapers and planes}\label{sec:examples-translator-jacobi}
The grim-reaper family was constructed in Proposition \ref{prop:grim-reaper-speed-family}. Here its arclength coordinate exposes the weighted stability operator and a distinction between strict positivity on compact tests and a positive spectral gap. The weight is the translator weight of \cite{HIMW19}, Proposition 3.2, with upward speed one.

\begin{proposition}[Global arclength coordinates for the unit grim reaper]\label{prop:grim-arclength-weight}
The curve $y=-\log\cos x$, $|x|<\pi/2$, has the global arclength parametrization
\[
 X(s)=(\arctan(\sinh s),\log\cosh s),\qquad s\in\mathbb R.
\]
With upward normal $\nu=(-\tanh s,\operatorname{sech}s)$, its scalar curvature is $H=-\operatorname{sech}s$. For velocity $v=e_2$, its weight and Jacobi operator are
\[
 \rho_v=\cosh s,\qquad J_v=\frac{d^2}{ds^2}+\tanh s\frac d{ds}+\operatorname{sech}^2s.
\]
\end{proposition}
\begin{proof}
The identity $\cos(\arctan(\sinh s))=\operatorname{sech}s$ places the point on the graph. Its derivative is $(\operatorname{sech}s,\tanh s)$ and has length one; the first coordinate increases onto $(-\pi/2,\pi/2)$. Differentiating the normal gives $\nu'=-\operatorname{sech}s\,X'$, proving the signed curvature. The height is $h=\log\cosh s$, whose derivative is $\tanh s$, and $|A|^2=\operatorname{sech}^2s$. Definition \ref{def:translator-jacobi-weight} gives the displayed operator.
\end{proof}

\begin{proposition}[The two translation fields and the weighted square-integrable kernel]\label{prop:grim-jacobi-kernel}
All smooth real solutions of $J_vu=0$ on the complete unit grim reaper are
\[
 u(s)=a\operatorname{sech}s+b\tanh s.
\]
Both basis functions are bounded, but the weighted square-integrable kernel is exactly the span of $w=\operatorname{sech}s$, with
\[
 \int_{\mathbb R}w^2\rho_v\,ds=\pi.
\]
\end{proposition}
\begin{proof}
The two functions, up to sign, are the normal components of vertical and horizontal translations. Proposition \ref{prop:translator-angle-jacobi} therefore makes them solutions; substitution in the displayed operator also verifies this directly. At $s=0$ their initial data are $(1,0)$ and $(0,1)$, so they form a basis of the two-dimensional solution space of the smooth second-order ODE. Their boundedness is immediate. If $b\ne0$, the function has a nonzero limit at either infinite end while $\rho_v=\cosh s$ grows exponentially, so it is not weighted square-integrable. For $w$, the integrand is $\operatorname{sech}s$. Since $(\arctan(\sinh s))'=\operatorname{sech}s$, its integral is $\pi$.
\end{proof}

\begin{proposition}[Strict compact stability with zero spectral bottom]\label{prop:grim-zero-spectral-bottom}
For every nonzero $\phi\in C_c^\infty(\mathbb R)$ the grim-reaper form satisfies
\[
 Q_v(\phi)=\int_{\mathbb R}\operatorname{sech}s\,
             |(\phi\cosh s)'|^2ds>0.
\]
Nevertheless its variational spectral bottom is
\[
 \inf_{0\ne\phi\in C_c^\infty(\mathbb R)}
 \frac{Q_v(\phi)}{\int\phi^2\cosh s\,ds}=0.
\]
\end{proposition}
\begin{proof}
Insert $w=\operatorname{sech}s$ in Proposition \ref{prop:translator-ground-state} to get the formula. Equality would make $\phi\cosh s$ constant; compact support forces that constant to be zero. To compute the infimum, choose a fixed smooth cutoff $\chi$ equal to one on $[-1,1]$, supported in $[-2,2]$, and put $\phi_R=w\chi(s/R)$. Then
\[
 Q_v(\phi_R)=R^{-2}\int\operatorname{sech}s\,|\chi'(s/R)|^2ds
 \le R^{-2}\|\chi'\|_\infty^2\pi,
\]
while $\int\phi_R^2\rho_v\to\pi$ by dominated convergence. Stability gives the opposite bound zero. The null function $w$ in the weighted square-integrable kernel is not compactly supported, so there is no contradiction with strict positivity on compact tests.
\end{proof}

\begin{proposition}[A one-dimensional Schrödinger form]\label{prop:grim-schrodinger-form}
The substitution $\psi=\sqrt{\cosh s}\,\phi$ preserves the weighted $L^2$ norm and transforms the grim-reaper quadratic form into
\[
 Q_v(\phi)=\int_{\mathbb R}\left[|\psi'|^2+
       \left(\frac14-\frac34\operatorname{sech}^2s\right)\psi^2\right]ds
\]
for compactly supported smooth real tests. The weighted null field $\operatorname{sech}s$ becomes $\operatorname{sech}^{1/2}s$.
\end{proposition}
\begin{proof}
Writing $h=\log\cosh s$ gives $\phi=e^{-h/2}\psi$. Thus the weighted derivative energy is $\int(\psi'-h'\psi/2)^2$. Integrate the cross term by parts; its contribution is $h''\psi^2/2$. The resulting potential for the full quadratic form is
$h''/2+(h')^2/4-|A|^2$. Substituting $h''=\operatorname{sech}^2s$, $(h')^2=1-\operatorname{sech}^2s$ and $|A|^2=\operatorname{sech}^2s$ gives the formula. The norm identity and the transformed null field follow immediately from the substitution. No complete spectral resolution is asserted.
\end{proof}

\begin{proposition}[Grim cylinders have a different square-integrable kernel]\label{prop:grim-cylinder-l2-kernel}
Let $M$ be the product of the complete unit grim reaper with $\mathbb R^k$, $k\ge1$, translating in its grim-reaper plane. It is weighted stable and has variational spectral bottom zero, but it has no nonzero smooth Jacobi field in $L^2(\rho_vdA)$.
\end{proposition}
\begin{proof}
The metric is $ds^2+|dy|^2$, the weight is $\cosh s$, and the operator is the one in Proposition \ref{prop:grim-arclength-weight} plus $\Delta_y$. The function $w(s,y)=\operatorname{sech}s$ is a positive Jacobi field, proving stability. It is not weighted square-integrable because its squared integral in $s$ is $\pi$ and the Euclidean factor has infinite volume. Lemma \ref{lem:translator-l2-kernel-ground-state} therefore gives the zero kernel.

For the spectral bottom, multiply the cutoff tests of Proposition \ref{prop:grim-zero-spectral-bottom} by $\eta(y/R)$, with nonzero smooth compactly supported $\eta$ on $\mathbb R^k$. Separation of variables makes the Rayleigh quotient the sum of the grim-reaper quotient and
$R^{-2}\int|\nabla\eta|^2/\int\eta^2$. Both tend to zero. This example has zero infimum without a nonzero weighted square-integrable null field; it does not require an assertion about the entire spectrum.
\end{proof}

\begin{example}[A translating plane has a positive weighted gradient gap]\label{ex:translator-plane-gap}
Let $P\cong\mathbb R^n$ be an affine hyperplane and let $0\ne v$ be tangent to it. The image is stationary under translation by $v$, but its translator-weighted form has variational spectral bottom $|v|^2/4$. Its induced Euclidean metric is complete and its induced conformal translator metric is incomplete.
\end{example}
\begin{proof}
Since $A=0$, the Jacobi form equals the weighted gradient energy. For $\psi=e^{\langle x,v\rangle/2}\phi$, integration of the compactly supported cross term gives
\[
 Q_v(\phi)=\int_P\left(|\nabla\psi|^2+\frac{|v|^2}{4}\psi^2\right)dA,
 \qquad \int_P\phi^2\rho_v\,dA=\int_P\psi^2dA.
\]
The lower bound is immediate. Taking $\psi_R(x)=\eta((x-p)/R)$ for a fixed nonzero compactly supported smooth $\eta$ makes its gradient Rayleigh quotient tend to zero, proving sharpness. Euclidean completeness is that of an affine plane. The ray $p-tv/|v|$ stays in $P$ and has finite conformal length by Proposition \ref{prop:translator-conformal-metric}, proving the final assertion. A tangential velocity changes this weighted functional even though it does not change the plane's image under the flow.
\end{proof}

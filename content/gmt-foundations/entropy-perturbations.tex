\chapter{Entropy-lowering compact perturbations}\label{sec:entropy-perturbations}
A negative second variation at one Gaussian center does not control the supremum over all centers and scales. We supply the additional estimates in \cite{CM12}, Theorem 0.15 and its proof in Section 7.3, pp.~789--792. Throughout, Gaussian scale is a positive parameter, not flow time. The absence of a Euclidean line factor is essential to this argument.

\begin{definition}[A compact normal graph family]\label{def:compact-shrinker-perturbation}
Let $\Sigma^n\subset\mathbb R^{n+1}$ be a smooth properly embedded two-sided hypersurface, and let $f\in C_c^\infty(\Sigma)$. For sufficiently small $|s|$, set
\[
 \Sigma_s=\{x+sf(x)\nu(x):x\in\Sigma\},\qquad
 G(c,t,s)=F_{c,t}(\Sigma_s).
\]
The graph is taken within a fixed tubular neighborhood of the compact support. A finite coordinate cover and compactness give an interval on which the maps remain embeddings. All changed points then lie in a fixed ambient compact set; their areas and curvatures are uniformly bounded on a smaller closed parameter interval. This is the variation class specified in \cite{CM12}, p.~763, footnote 6. The unchanged part may have unbounded curvature.
\end{definition}

\begin{lemma}[A Gaussian scale derivative identity and inequality]\label{lem:gaussian-scale-inequality}
Let $\Sigma^n\subset\mathbb R^{n+1}$ be a smooth properly immersed two-sided boundaryless hypersurface with polynomial area growth and with $|H(x)|\leq C(1+|x|)^d$ for some finite $C,d$. For every $c$ and $t>0$,
\[
 \partial_t F_{c,t}(\Sigma)
 =\int_\Sigma\left[\left(\frac{\langle x-c,\nu\rangle}{2t}-\frac H2\right)^2
 -\frac{H^2}{4}\right]\rho_{c,t}\,dA
 \geq-\frac14\int_\Sigma H^2\rho_{c,t}\,dA.
\]
\end{lemma}
\begin{proof}
Write $z=x-c$ and $\rho=\rho_{c,t}$. Kernel differentiation and $\Delta x=-H\nu$ give
\[
 \partial_t\rho=\rho\left(\frac{|z|^2}{4t^2}-\frac n{2t}\right),\qquad
 \Delta_\Sigma\rho=\rho\left(\frac{|z^T|^2}{4t^2}-\frac n{2t}
 +\frac{H\langle z,\nu\rangle}{2t}\right).
\]
Their difference is the integrand in the stated identity. The integral of $\Delta_\Sigma\rho$ is zero: multiply by compact radial cutoffs, integrate by parts, and bound the error by the Gaussian tail of $C(1+|x|)\rho$. Properness and polynomial growth justify this limit; the polynomial bound for $H$ also makes $\Delta_\Sigma\rho$ integrable. Differentiation under the integral is justified by Gaussian moments. Dropping the nonnegative square gives the inequality. This is the square completion behind \cite{CM12}, (7.3)--(7.5).
\end{proof}

\begin{lemma}[Uniform bounds for the changed compact part]\label{lem:entropy-compact-perturbation-bounds}
Let $\Sigma_s$ be a compact normal graph family over a smooth proper embedded hypersurface of polynomial area growth. On a sufficiently small fixed closed interval of $s$, there are constants $A_0,C_0$ such that
\[
 G(c,t,s)\leq F_{c,t}(\Sigma)+A_0(4\pi t)^{-n/2}
 \exp\left(-\frac{\operatorname{dist}(c,B_{C_0}(0))^2}{4t}\right).
\]
Consequently the added term is at most $A_0(4\pi t)^{-n/2}$, and for $|c|>C_0+R$ it is at most $C(n)A_0R^{-n}$ uniformly for all $t>0$. On each compact set of $(c,t)$, $G$ and its derivatives in $s$ are continuous, and $G(c,t,s)\to F_{c,t}(\Sigma)$ uniformly as $s\to0$.
\end{lemma}
\begin{proof}
Choose a compact region in the parameter surface whose interior contains the support of $f$. Outside this region all embeddings are identical. Its perturbed image lies in $B_{C_0}$ and has area at most $A_0$, uniformly in $s$. Subtract the integrals on the common unchanged region and discard the nonnegative old contribution from the compact region to obtain the inequality.

For far centers the exponential is at most $e^{-R^2/(4t)}$. The substitution $t=R^2u$ gives
\[
 \sup_{t>0}t^{-n/2}e^{-R^2/(4t)}
 =R^{-n}\sup_{u>0}u^{-n/2}e^{-1/(4u)}=C_nR^{-n}.
\]
The last supremum is finite since $n\geq1$ and the function tends to zero at both ends. Shape derivatives involve only a fixed compact region and smooth bounded geometry. The unchanged part is controlled by polynomial Gaussian moments for parameters in compact positive-scale sets. Differentiation and uniform continuity therefore give the last assertions. Compare \cite{CM12}, (7.40)--(7.46).
\end{proof}

\begin{lemma}[Small-scale control for compact perturbations of a shrinker]\label{lem:entropy-small-scale-control}
Let $\Sigma_s$ be a compact normal graph family over a connected complete smooth properly embedded two-sided boundaryless self-shrinker $\Sigma$ with polynomial area growth. Fix $R<\infty$. There are constants $A,B,C\geq0$, independent of small $s$, $|c|\leq R$, and $0<t\leq1$, such that
\[
 \partial_tG(c,t,s)\geq-A-BG(c,t,s).
\]
In particular, for $0<t\leq\tau\leq1$,
\[
 G(c,t,s)\leq e^{B\tau}G(c,\tau,s)+C\tau.
\]
\end{lemma}
\begin{proof}
On the unchanged part of the surface, $|H|^2=|x^\perp|^2/4\leq|x|^2/4$. On the changed compact region the curvatures are uniformly bounded by some $C_H$. Each perturbed surface has polynomial area and curvature growth, so Lemma \ref{lem:gaussian-scale-inequality} applies and gives
\[
 \partial_tG\geq-\frac1{16}\int_\Sigma|x|^2\rho_{c,t}\,dA
 -\frac{C_H^2}{4}G.
\]
Theorem \ref{thm:shrinker-entropy-maximum} gives finite entropy of $\Sigma$. Proposition \ref{prop:entropy-gaussian-second-moment} bounds the first integral by $2\lambda(\Sigma)(R^2+C_nt)$, proving the differential inequality with uniform constants. Multiply by $e^{Bt}$ and integrate from $t$ to $\tau$ to obtain
\[
 G(c,t,s)\leq e^{B(\tau-t)}G(c,\tau,s)
 +A e^{-Bt}\int_t^\tau e^{Bu}\,du
 \leq e^{B\tau}G(c,\tau,s)+Ae^B\tau.
\]
This proves the claim with $C=Ae^B$. It is the uniform small-scale estimate in \cite{CM12}, (7.49)--(7.51); no curvature bound at infinity is assumed for the original shrinker.
\end{proof}

\begin{lemma}[Maximizing parameters cannot escape under a small compact perturbation]\label{lem:entropy-parameter-exclusion}
Let $\Sigma_s$ satisfy the hypotheses of Lemma \ref{lem:entropy-small-scale-control}, and suppose the constant tangent subspace of $\Sigma$ is zero. There exist $R,T>1$, $0<\tau<1$, $d>0$, and $\eta>0$ such that
\[
 G(c,t,s)\leq\lambda(\Sigma)-d
\]
for $|s|\leq\eta$ whenever $|c|>R$, or $t>T$, or $t<\tau$.
\end{lemma}
\begin{proof}
Theorem \ref{thm:shrinker-global-entropy-gap} gives a fixed positive deficit for the unperturbed surface whenever $|c|$ is large or $t\notin[1/2,2]$. Choose a smaller positive $\delta$ so that this deficit is at least $4\delta$ and $4\delta<\lambda(\Sigma)$. The far-center estimate of Lemma \ref{lem:entropy-compact-perturbation-bounds} then gives deficit at least $3\delta$ for every scale when $|c|>R$, after increasing $R$. Its large-scale estimate similarly gives deficit at least $3\delta$ for all centers when $t>T$. These choices are uniform on a fixed small shape interval.

It remains to consider $|c|\leq R$ at small scales. Take $A,B,C$ from Lemma \ref{lem:entropy-small-scale-control}. Choose $0<\tau<1/2$ sufficiently small that
\[
 e^{B\tau}(\lambda(\Sigma)-3\delta)+C\tau
 \leq\lambda(\Sigma)-2\delta.
\]
At this fixed scale, uniform convergence on the compact center ball makes $G(c,\tau,s)\leq\lambda(\Sigma)-3\delta$ for $|s|\leq\eta$, after reducing $\eta$, because the unperturbed deficit is at least $4\delta$. The small-scale comparison propagates this upper bound to all $0<t<\tau$ with deficit $2\delta$. Combining the three regions proves the claim, for instance with $d=2\delta$. This is the escape-prevention part of \cite{CM12}, pp.~790--792.
\end{proof}

\begin{lemma}[An F-negative direction gives a strict joint local maximum]\label{lem:entropy-joint-local-maximum}
Let $\Sigma$ be a connected complete smooth properly embedded two-sided boundaryless self-shrinker with polynomial area growth and zero constant tangent subspace. Suppose $f\in C_c^\infty(\Sigma)$ makes the joint Hessian in Theorem \ref{thm:shrinker-joint-hessian} negative for every center and scale derivative. For its compact graph family there are $r,\gamma>0$ such that
\[
 G(c,t,s)\leq\lambda(\Sigma)
 -\gamma\bigl(|c|^2+(t-1)^2+s^2\bigr)
\]
whenever $|c|^2+(t-1)^2+s^2<r^2$.
\end{lemma}
\begin{proof}
All first derivatives vanish at $(0,1,0)$ by Proposition \ref{prop:gaussian-joint-first}. Consider a Hessian direction $(y,h,b)$ in center, scale, and graph amplitude. If $b\ne0$, its value is $b^2$ times the joint Hessian for shape speed $f$ with center/scale derivatives $y/b,h/b$, so it is negative. If $b=0$ and $(y,h)\ne0$, it is negative by the argument of Proposition \ref{prop:shrinker-local-entropy-gap}. Thus the full finite-dimensional Hessian is negative definite. Smooth dependence of $G$ on these parameters follows from compact shape support and Gaussian domination of the unchanged part. Taylor's theorem and a uniform bound on the unit sphere of directions give the estimate. This is \cite{CM12}, step (1) on p.~790.
\end{proof}

\begin{theorem}[F-instability lowers entropy when no line splits]\label{thm:shrinker-entropy-lowering}
Let $\Sigma^n\subset\mathbb R^{n+1}$, $n\geq1$, be connected, complete, smooth, properly embedded, two-sided, without boundary, and a self-shrinker with polynomial area growth. Suppose it has no nonzero constant tangent vector and is $F$-unstable in Definition \ref{def:shrinker-f-stability}. Then there exist $f\in C_c^\infty(\Sigma)$ and $\eta>0$ such that the normal graphs $\Sigma_s=\{x+sf(x)\nu(x):x\in\Sigma\}$ satisfy
\[
 \lambda(\Sigma_s)<\lambda(\Sigma)\qquad(0<|s|<\eta).
\]
These perturbations can be arbitrarily small in every fixed $C^m$ norm. This is a proper embedded formulation of \cite{CM12}, Theorem 0.15, p.~763, proved on pp.~789--792.
\end{theorem}
\begin{proof}
Choose $f$ from $F$-instability. Lemma \ref{lem:entropy-joint-local-maximum} supplies a joint quadratic deficit near $(0,1,0)$. Lemma \ref{lem:entropy-parameter-exclusion} supplies a compact parameter box $K=\{|c|\leq R,\ \tau\leq t\leq T\}$ outside which all sufficiently small perturbations have a fixed deficit $d>0$.

Choose a parameter neighborhood $U$ of $(0,1)$ with closure in $K$, sufficiently small that the local quadratic estimate holds on $U$ for all sufficiently small $s$. On the compact set $K\setminus U$, the unperturbed surface has a uniform positive deficit by Theorem \ref{thm:shrinker-global-entropy-gap}. Uniform convergence of $G(c,t,s)$ to $G(c,t,0)$ there retains a fixed positive deficit $d_1$ after reducing $|s|$. On $U$ the quadratic estimate gives deficit at least $\gamma s^2$. Consequently, for each small nonzero $s$,
\[
 \sup_{c,\ t>0}G(c,t,s)
 \leq\lambda(\Sigma)-\min\{d,d_1,\gamma s^2\}
 <\lambda(\Sigma).
\]
The graph family remains embedded for small amplitude by Definition \ref{def:compact-shrinker-perturbation}. Since $f$ is fixed, smooth and compactly supported, the graph function $sf$ tends to zero in each fixed $C^m$ norm. The exclusion of escaping parameters is what permits taking the global supremum.
\end{proof}

\begin{corollary}[Entropy-lowering perturbations of closed sign-changing shrinkers]\label{cor:closed-shrinker-entropy-drop}
A connected smooth closed embedded two-sided self-shrinker whose scalar mean curvature changes sign admits arbitrarily small compact normal perturbations with strictly smaller entropy.
\end{corollary}
\begin{proof}
Theorem \ref{thm:shrinker-changing-sign-instability} gives an $F$-negative direction, using the compact elliptic principal-eigenfunction theorem recorded there. A closed hypersurface cannot split a Euclidean line, since its image is bounded; Proposition \ref{prop:shrinker-translation-splitting} therefore makes its constant tangent subspace zero. All other hypotheses of Theorem \ref{thm:shrinker-entropy-lowering} follow from closed smooth embeddedness. Applying it proves the result. No classification of mean-convex shrinkers is used, and this corollary is not stated for all nonround shrinkers without that additional input.
\end{proof}

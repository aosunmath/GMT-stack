\chapter{Fiber integration and selecting useful levels}\label{sec:coarea-selection-disintegration}
The coarea identity has consequences beyond a single integral formula. It identifies an absolutely continuous pushforward after weighting by the Jacobian, controls the regular part after dividing by the Jacobian, and permits simultaneous choices of levels. We expand \cite{Sim18}, Chapter 2, Theorem 7.5 and Remarks 7.7, printed pp.~80--81, and Chapter 3, equation (2.9), p.~93. Throughout $M\subset U\subset\mathbb R^n$ is measurable, countably $m$-rectifiable and locally finite in $\mathcal H^m$, $U$ is open, $f:U\to\mathbb R^k$ is locally Lipschitz, and $1\leq k<m$. Write $J=J_k(d^Mf)$ and $M_y=M\cap f^{-1}(y)$.

\begin{proposition}[The pushforward of Jacobian-weighted area]\label{prop:coarea-weighted-pushforward}
For every nonnegative measurable $h$ on $M$, the Borel pushforward satisfies
\[
 f_\#(hJ\mathcal H^m\llcorner M)=F\mathcal L^k,
 \qquad F(y)=\int_{M_y}h\,d\mathcal H^{m-k}
 \quad\text{almost everywhere}.
\]
The fiber integral has a measurable representative. Both measures may have infinite mass; this formula does not itself assert local finiteness of the pushforward.
\end{proposition}
\begin{proof}
Choose Borel representatives modulo the relevant Hausdorff-null sets, using the representative and null-fiber steps in Theorem \ref{thm:rectifiable-coarea-chart-proof}. Apply that theorem to the weight $h\chi_{f^{-1}(B)}$, where $B\subset\mathbb R^k$ is Borel. On the fiber at $y$ the extra indicator is exactly $\chi_B(y)$, so the identity becomes
\[
 \int_{M\cap f^{-1}(B)}hJ\,d\mathcal H^m
 =\int_BF(y)\,dy.
\]
This is equality of the indicated Borel measures. The choices of representatives do not alter either side: their differences lie in a Hausdorff-null set and almost every fiber meets that set in lower-dimensional Hausdorff measure zero. The convention $0\cdot\infty=0$ is inherited from coarea. Global finite mass, when imposed, makes the density integrable; it is not required for the equality.
\end{proof}

\begin{proposition}[Dividing by the Jacobian on the regular part]\label{prop:coarea-inverse-jacobian}
Put $R=\{x\in M:J(x)>0\}$ and $C=M\setminus R$, with $J=0$ on its exceptional undefined set. For any nonnegative measurable $a$,
\[
 f_\#(a\mathcal H^m\llcorner R)=G\mathcal L^k,
 \qquad G(y)=\int_{R\cap M_y}\frac{a}{J}\,d\mathcal H^{m-k}.
\]
Consequently
\[
 f_\#(a\mathcal H^m\llcorner M)=G\mathcal L^k+f_\#(a\mathcal H^m\llcorner C).
\]
The second summand is a critical-set contribution; it is not asserted to be singular with respect to Lebesgue measure. If $\int_Ra\,d\mathcal H^m<\infty$, then $G\in L^1(\mathbb R^k)$ and is finite almost everywhere.
\end{proposition}
\begin{proof}
Use Proposition \ref{prop:coarea-weighted-pushforward} with $h=a/J$ on $R$ and zero on $C$. This is a nonnegative measurable weight, and $hJ=a\chi_R$ with the stated zero convention. The resulting density is exactly $G$. Add the pushforward of the complementary measure to obtain the second formula. The total integral of $G$ equals the regular mass by the same identity, proving the last assertion. Although the critical set has zero lower-dimensional measure in almost every fiber by coarea, its $m$-dimensional measure can be positive. Thus its unweighted pushforward cannot simply be deleted. Example \ref{ex:coarea-flat-plateau-atom} gives a nonzero critical contribution explicitly.
\end{proof}

\begin{theorem}[The fiberwise Sard conclusion at C1 regularity]\label{thm:c1-sard-fiberwise}
Let $N\subset U$ be an embedded $m$-dimensional $C^1$ manifold and let $f|N$ be $C^1$. Put $C=\{x\in N:\operatorname{rank}d^Nf_x<k\}$. For almost every $y$, $N\cap f^{-1}(y)$ is the union of a possibly empty $(m-k)$-dimensional $C^1$ manifold and a relatively closed subset of that fiber with zero $\mathcal H^{m-k}$ measure. The manifold is its regular part, and its tangent at $x$ is $\ker d^Nf_x$. This is \cite{Sim18}, Chapter 2, Theorem 7.5, p.~80.
\end{theorem}
\begin{proof}
The rank-deficient set $C$ is relatively closed in $N$ because the tangential derivative is continuous and all $k\times k$ minors vanish on it. At a point of $N\setminus C$, the implicit function theorem makes the level set a $C^1$ manifold of dimension $m-k$, with tangent equal to the kernel. These local descriptions form the regular part of the level set as an embedded manifold.

Cover $N$ by the finite-area patches of Lemma \ref{lem:finite-area-c1-patches}. On each patch apply Theorem \ref{thm:rectifiable-coarea-chart-proof} to the indicator of $C$. Its tangential Jacobian is zero, so the integral of the critical fiber measure is zero. This measure is therefore zero at almost every level. The countable union of exceptional level sets is null, proving the assertion on all of $N$, without assuming the entire nonproper manifold has locally finite ambient area. The result says that the critical part of almost every fiber is negligible; it does not say that this part is empty. The stronger critical-value conclusion requires additional smoothness hypotheses, as distinguished in \cite{Sim18}, Remark 7.6.
\end{proof}

\begin{proposition}[Selecting levels that satisfy finitely many bounds]\label{prop:coarea-simultaneous-level-selection}
Let $Y\subset\mathbb R^k$ be measurable with $0<v=\mathcal L^k(Y)<\infty$. Let $h_1,\ldots,h_s$ be nonnegative measurable weights with
\[
 E_i=\int_{M\cap f^{-1}(Y)}h_iJ\,d\mathcal H^m<\infty.
\]
Choose $\varepsilon_i>0$ with $\sum_i\varepsilon_i<1$. Outside any prescribed null set of levels, there is a measurable set $S\subset Y$ of measure at least $(1-\sum_i\varepsilon_i)v$ on which all the estimates
\[
 \int_{M_y}h_i\,d\mathcal H^{m-k}\leq\frac{E_i}{\varepsilon_i v}
 \quad(i=1,\ldots,s)
\]
hold. If $E_i=0$, the corresponding bound is zero.
\end{proposition}
\begin{proof}
Coarea makes each fiber cost $F_i(y)$ measurable, with $\int_YF_i=E_i$. If $E_i>0$, Markov's inequality gives measure at most $\varepsilon_i v$ to the set $\{F_i>E_i/(\varepsilon_i v)\}$. If $E_i=0$, nonnegativity gives $F_i=0$ almost everywhere, so its positive set is null. Remove the finite union of these bad sets and the prescribed null set. Subadditivity gives the asserted measure lower bound, and every remaining point satisfies every inequality. The strict positivity of this lower bound ensures that an admissible level exists. This quantifies the good-level principle of \cite{Sim18}, Remark 7.7(1), and makes explicit how several requirements can be imposed at once.
\end{proof}

\begin{proposition}[Good levels along a level-dependent subsequence]\label{prop:coarea-subsequence-good-levels}
Let $Y$ and $v$ be as above, and let $F_j:Y\to[0,\infty]$ be measurable with $E_j=\int_YF_j<\infty$. For any $0<\varepsilon<1$, there is a measurable set $S\subset Y$ of measure at least $(1-\varepsilon)v$ such that for each $y\in S$ an infinite subsequence $j_l=j_l(y)$ satisfies
\[
 F_{j_l}(y)\leq\frac{E_{j_l}}{\varepsilon v}.
\]
In coarea applications, $F_j$ may be the weighted fiber costs for different rectifiable sets. The subsequence is allowed to depend on $y$. This is the selection content of \cite{Sim18}, Remark 7.7(2), pp.~80--81.
\end{proposition}
\begin{proof}
After removing the countable union of null sets associated with any zero $E_j$, define $G_j=\{F_j\leq E_j/(\varepsilon v)\}$. Markov's inequality gives $|G_j|\geq(1-\varepsilon)v$. For each $N$, the union $H_N=\bigcup_{j\geq N}G_j$ therefore has at least this measure. The sets $H_N$ decrease and lie in the finite-measure set $Y$, so continuity from above gives $|\bigcap_NH_N|\geq(1-\varepsilon)v$. This intersection is the limsup of the good sets: each of its points lies in infinitely many $G_j$. Listing those indices in increasing order gives the required subsequence. Nothing in this argument selects one subsequence that works for every level simultaneously.
\end{proof}

\begin{proposition}[Summable errors vanish on almost every fiber]\label{prop:coarea-summable-fiber-errors}
Let $F_j:Y\to[0,\infty]$ be measurable with $\sum_j\int_YF_j<\infty$. Then $\sum_jF_j(y)<\infty$ and $F_j(y)\to0$ for almost every $y$. In particular, if coarea gives fiber costs with $\int_YF_j\to0$, one can choose a subsequence of indices, independent of the level, along which the costs tend to zero almost everywhere.
\end{proposition}
\begin{proof}
Tonelli's theorem gives $\int_Y\sum_jF_j=\sum_j\int_YF_j<\infty$. The nonnegative sum is finite almost everywhere, and its terms tend to zero there. For the last assertion choose increasing indices $j_l$ with $\int_YF_{j_l}\leq2^{-l}$; the first assertion then applies to this subsequence. Here the conclusion is almost-everywhere convergence along one common subsequence, under the stronger vanishing-integral hypothesis. It should not be confused with the bounded normalized-cost subsequences in Proposition \ref{prop:coarea-subsequence-good-levels}.
\end{proof}

\begin{proposition}[The distance-volume function is absolutely continuous off its critical part]\label{prop:rectifiable-distance-volume-derivative}
Let $a\in\mathbb R^n$ and $0<\alpha<\beta$ with $\overline{B_\beta(a)}\subset U$. Put $\rho(x)=|x-a|$, $J=|\nabla^M\rho|$, $R=\{x\in M:J(x)>0\}$, and
\[
 V_R(t)=\mathcal H^m(R\cap\{\alpha<\rho<t\}),\qquad \alpha\leq t\leq\beta.
\]
Then $V_R$ is absolutely continuous and for almost every $t$,
\[
 V_R'(t)=\int_{R\cap\partial B_t(a)}\frac1{|\nabla^M\rho|}\,d\mathcal H^{m-1}
 \geq\mathcal H^{m-1}(M\cap\partial B_t(a)).
\]
The inequality holds at almost every $t\in(\alpha,\beta)$; coarea discards the critical part of the fiber at these levels. This statement concerns the regular part of the distance function and does not assert absolute continuity of the full volume function.
\end{proposition}
\begin{proof}
The restricted area in the compact annulus is finite. Proposition \ref{prop:coarea-inverse-jacobian}, applied to $f=\rho$ and its regular part, gives an integrable density
\[
 G(t)=\int_{R\cap\partial B_t(a)}J^{-1}\,d\mathcal H^{m-1}
\]
on $(\alpha,\beta)$. Thus $V_R(t)=\int_\alpha^tG(s)\,ds$. The fundamental theorem for Lebesgue integrals proves absolute continuity and the derivative formula. Since $|\nabla^M\rho|\leq1$, its reciprocal on $R$ is at least one. The critical-fiber measure vanishes for almost every radius by coarea with the indicator of $\{J=0\}$; this proves the final comparison with the whole fiber. A surface lying on one sphere has a nonzero critical contribution and a jump in the full volume function, as illustrated in Example \ref{ex:coarea-spherical-critical-level}.
\end{proof}
